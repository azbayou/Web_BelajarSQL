-- 1. DDL (Data Definition Language)
-- Mengatur struktur database & tabel

-- Membuat Tabel Baru
CREATE TABLE employees (
    id INT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    salary DECIMAL(10, 2),
    hire_date DATE
);

-- Menghapus Tabel
DROP TABLE employees;

-- Menambah Kolom Baru
ALTER TABLE employees ADD department VARCHAR(50);


-- 2. DML (Data Manipulation Language)
-- Memanipulasi data di dalam tabel

-- Menambahkan Data (Insert)
INSERT INTO employees (id, name, salary, hire_date) 
VALUES (1, 'Andi', 5000.00, '2023-01-15'),
       (2, 'Budi', 7000.50, '2022-05-20');

-- Mengubah Data (Update)
UPDATE employees 
SET salary = 5500.00 
WHERE id = 1;

-- Menghapus Data (Delete)
DELETE FROM employees 
WHERE id = 2;


-- 3. DQL (Data Query Language)
-- Menampilkan & Mencari Data (SELECT)

-- Mengambil semua kolom
SELECT * FROM employees;

-- Mengambil kolom tertentu
SELECT name, salary FROM employees;

-- Menggunakan kondisi (WHERE)
SELECT * FROM employees WHERE salary > 5000;

-- Mengurutkan data (ORDER BY)
SELECT * FROM employees ORDER BY salary DESC; -- DESC (Besar ke kecil), ASC (Kecil ke besar)

-- Membatasi jumlah hasil (LIMIT)
SELECT * FROM employees LIMIT 10;


-- 4. AGREGASI & GROUPING
-- Menghitung total, rata-rata, dll

-- Menghitung jumlah baris
SELECT COUNT(*) FROM employees;

-- Mencari total, rata-rata, nilai tertinggi & terendah
SELECT SUM(salary), AVG(salary), MAX(salary), MIN(salary) FROM employees;

-- Mengelompokkan data (GROUP BY)
SELECT department, COUNT(*) as total_karyawan, AVG(salary) as rata_gaji
FROM employees
GROUP BY department;

-- Kondisi setelah agregasi (HAVING)
SELECT department, AVG(salary) 
FROM employees 
GROUP BY department 
HAVING AVG(salary) > 6000;


-- 5. JOIN TABEL
-- Menggabungkan data dari lebih dari 1 tabel

-- INNER JOIN (Hanya data yang cocok di kedua tabel)
SELECT e.name, d.department_name
FROM employees e
INNER JOIN departments d ON e.department_id = d.id;

-- LEFT JOIN (Semua data dari tabel kiri + data yang cocok dari kanan)
SELECT e.name, d.department_name
FROM employees e
LEFT JOIN departments d ON e.department_id = d.id;


-- 6. FUNGSI STRING & TANGGAL (Sering dipakai)
-- Manipulasi teks dan waktu

-- Menggabungkan teks
SELECT CONCAT(first_name, ' ', last_name) AS full_name FROM users;

-- Ekstrak Tahun dari Tanggal
SELECT EXTRACT(YEAR FROM hire_date) FROM employees;

-- Format Tanggal menjadi string (Beda-beda tiap database, ini standar PostgreSQL/DuckDB)
SELECT strftime(hire_date, '%Y-%m-%d') FROM employees;


-- 7. SUBQUERY & CTE (Common Table Expression)

-- Subquery (Query di dalam query)
SELECT name, salary 
FROM employees 
WHERE salary > (SELECT AVG(salary) FROM employees);

-- CTE (Membuat tabel sementara sebelum query utama)
WITH HighEarners AS (
    SELECT name, salary FROM employees WHERE salary > 10000
)
SELECT * FROM HighEarners;
